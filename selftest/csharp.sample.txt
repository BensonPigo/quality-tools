namespace Shop;

public enum MemberLevel { NonMember, Member, VIP }

public record Item(int Price, int Qty, bool OnSale);

public static class Cart
{
    // 故意寫亂的函式：所有規則都塞在同一個函式裡。
    public static int Checkout(IReadOnlyList<Item> items, MemberLevel level, string coupon)
    {
        if (items.Count == 0)
        {
            return 0;
        }

        var total = 0;
        foreach (var it in items)
        {
            if (it.Qty <= 0)
            {
                continue;
            }
            var line = it.Price * it.Qty;
            if (!it.OnSale)
            {
                if (level == MemberLevel.Member)
                {
                    line = line * 90 / 100;
                }
                else if (level == MemberLevel.VIP)
                {
                    line = line * 85 / 100;
                }
            }
            total += line;
        }

        if (coupon == "SAVE100" && total >= 1000)
        {
            total -= 100;
        }
        if (total >= 3000)
        {
            total -= 300;
        }

        var shipping = 0;
        if (total < 1000 && level != MemberLevel.VIP && coupon != "FREESHIP")
        {
            shipping = 100;
            if (coupon == "HALFSHIP")
            {
                shipping = 50;
            }
        }
        return total + shipping;
    }
}

namespace Shop.Shipping;

public static class Fees
{
    // 依金額和地區算運費。
    public static int Fee(int total, string region)
    {
        if (total >= 1000)
        {
            return 0;
        }
        if (region == "離島")
        {
            return 150;
        }
        return 60;
    }

    // 依重量加價。
    public static int Weight(int kg)
    {
        if (kg > 20)
        {
            return 200;
        }
        if (kg > 10)
        {
            return 100;
        }
        if (kg > 5)
        {
            return 50;
        }
        if (kg > 2)
        {
            return 20;
        }
        return 0;
    }

    public static int Double(int x) => x * 2;
}

using Shop;
using Shop.Shipping;

public class CartTests
{
    // 每個正確答案都是依照需求規格親手算出的寫死數字。
    [Fact]
    public void 空購物車() => Assert.Equal(0, Cart.Checkout(new List<Item>(), MemberLevel.NonMember, ""));

    [Fact]
    public void 非會員_500元_加運費100() =>
        Assert.Equal(600, Cart.Checkout(new List<Item> { new(500, 1, false) }, MemberLevel.NonMember, ""));

    [Fact]
    public void 滿千免運() => Assert.Equal(0, Fees.Fee(1000, "本島"));
}
